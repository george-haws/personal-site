# Edge Analytics Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Record every page view of ghaws.dev at the Cloudflare edge and send it to a swappable analytics provider, starting with Umami Cloud, so visits survive ad blockers and tagged links identify where a visitor came from.

**Architecture:** A Cloudflare Worker runs in front of the static assets for every path outside `/_next/`. It serves the asset unchanged, then reports document loads (and the resume PDF) to a provider in the background. A small client component reports client-side route changes to the same Worker through a same-origin POST to `/pv`. The provider is one function behind a factory keyed by a config variable; the first implementation posts to Umami Cloud's events API.

**Tech Stack:** Cloudflare Workers with static assets (wrangler 4.149), TypeScript, Next.js 16.4 App Router static export, React 19, vitest 5 with Testing Library, isbot 5.

**Spec:** `docs/superpowers/specs/2026-10-11-edge-analytics-design.md`

## Global Constraints

- Node 24 (`.nvmrc`). If `node` fails with a `dyld ... libicui18n` error, the Homebrew node on the PATH is broken; run `source ~/.nvm/nvm.sh && nvm use` first in that shell.
- Invariant: analytics can never break the site. The asset response is obtained before tracking code runs; every line of tracking code sits inside a try/catch that logs with `console.error` and drops; the response is returned unchanged.
- Providers never throw. They log with `console.error` and resolve.
- Worker code under `worker/` never imports from `src/`, and `src/` never imports from `worker/`.
- The CSP in `public/_headers` is not changed.
- Pageview record shape (vendor-neutral), used everywhere:
  ```ts
  type PageView = {
    url: string;               // full URL including the query string
    hostname: string;
    referrer: string | null;
    userAgent: string;
    ip: string;                // CF-Connecting-IP
    language: string | null;   // first Accept-Language tag without its weight
  };
  ```
- Config variables in `wrangler.jsonc`: `ANALYTICS_PROVIDER` (`"umami"` or `"none"`), `UMAMI_WEBSITE_ID`, optional `UMAMI_ENDPOINT` (default `https://gateway.umami.is/api/send`).
- Routing: `run_worker_first: ["/*", "!/_next/*"]`. Ping path: `/pv`.
- Trackable document load: GET; asset status 200 or 304; `Sec-Fetch-Dest: document` (or, when that header is absent, `Accept` includes `text/html`) or path `/resume.pdf`; no `Sec-Purpose`/`Purpose` header containing `prefetch`; user agent not flagged by isbot. Paths under `/_next/` are never tracked.
- Commit messages follow the repo style: an imperative sentence as subject, a short body, ending with `Co-Authored-By: Claude Fable 5.1 <noreply@anthropic.com>`.
- `npm run check` (typecheck, lint, tests) must pass at the end of every task.
- Before editing anything under `src/`, read the relevant Next.js guide in `node_modules/next/dist/docs/` (named per task); this Next version differs from training data.

```ts
import { umamiProvider } from "./providers/umami";

/** One page view, vendor-neutral. Country is left to the vendor, which derives it from the IP. */
export type PageView = {
  /** Full URL including the query string, so UTM tags survive. */
  url: string;
  hostname: string;
  referrer: string | null;
  userAgent: string;
  /** CF-Connecting-IP. */
  ip: string;
  /** First Accept-Language tag without its weight, e.g. "en-US". */
  language: string | null;
};

/** Sends one page view somewhere. Implementations never throw: they log and resolve. */
export type Provider = { track(view: PageView): Promise<void> };

export type ProviderEnv = {
  ANALYTICS_PROVIDER?: string;
  UMAMI_WEBSITE_ID?: string;
  UMAMI_ENDPOINT?: string;
};

export const noopProvider: Provider = {
  async track() {},
};

const warned = new Set<string>();

function warnOnce(message: string) {
  if (warned.has(message)) return;
  warned.add(message);
  console.warn(message);
}

/** Picks the provider named by ANALYTICS_PROVIDER. Unset, "none", or anything unknown disables tracking. */
export function createProvider(env: ProviderEnv): Provider {
  const name = env.ANALYTICS_PROVIDER ?? "none";
  switch (name) {
    case "none":
      return noopProvider;
    case "umami": {
      const websiteId = env.UMAMI_WEBSITE_ID ?? "";
      if (websiteId === "") {
        warnOnce("analytics: UMAMI_WEBSITE_ID is empty; tracking disabled");
        return noopProvider;
      }
      return umamiProvider({ websiteId, endpoint: env.UMAMI_ENDPOINT });
    }
    default:
      warnOnce(`analytics: unknown ANALYTICS_PROVIDER "${name}"; tracking disabled`);
      return noopProvider;
  }
}
```

```ts
import type { PageView, Provider } from "../provider";

export const DEFAULT_UMAMI_ENDPOINT = "https://gateway.umami.is/api/send";

/** Cloudflare lets background work run after the response, but a hung vendor should not hold the Worker open. */
const TIMEOUT_MS = 5000;

export type UmamiOptions = { websiteId: string; endpoint?: string };

/**
 * Sends page views to Umami's events API. Umami derives country, browser,
 * device, UTM tags, and unique visitors from the fields sent here.
 */
export function umamiProvider(options: UmamiOptions): Provider {
  const endpoint = options.endpoint ?? DEFAULT_UMAMI_ENDPOINT;
  return {
    async track(view: PageView) {
      try {
        const url = new URL(view.url);
        const response = await fetch(endpoint, {
          method: "POST",
          headers: {
            "Content-Type": "application/json",
            // Umami rejects requests without a User-Agent; the visitor's is the one that matters.
            "User-Agent": view.userAgent,
          },
          body: JSON.stringify({
            type: "event",
            payload: {
              website: options.websiteId,
              hostname: view.hostname,
              url: url.pathname + url.search,
              referrer: view.referrer ?? "",
              // JSON.stringify drops undefined, so a null language is omitted rather than sent as null.
              language: view.language ?? undefined,
              ip: view.ip,
              userAgent: view.userAgent,
            },
          }),
          signal: AbortSignal.timeout(TIMEOUT_MS),
        });
        if (!response.ok) {
          console.error(`analytics: umami responded ${response.status}: ${await response.text()}`);
        }
      } catch (error) {
        console.error("analytics: umami request failed", error);
      }
    },
  };
}
```

```ts
import { createProvider, type PageView, type Provider } from "./provider";

/** First Accept-Language tag without its weight: "en-US,en;q=0.9" → "en-US". */
export function firstLanguage(header: string | null): string | null {
  if (!header) return null;
  const first = header.split(",")[0]?.split(";")[0]?.trim();
  return first ? first : null;
}

function isDocumentRequest(request: Request): boolean {
  const dest = request.headers.get("Sec-Fetch-Dest");
  if (dest !== null) return dest === "document";
  // Old browsers and some tools send no fetch metadata; fall back to content negotiation.
  return (request.headers.get("Accept") ?? "").includes("text/html");
}

/** Chrome prerender and search-engine prefetch send document requests nobody has looked at. */
function isPrefetch(request: Request): boolean {
  return ["Sec-Purpose", "Purpose"].some((name) =>
    (request.headers.get(name) ?? "").toLowerCase().includes("prefetch"),
  );
}

function isBot(request: Request): boolean {
  return isbot(request.headers.get("User-Agent") ?? "");
}

/** Whether a request that was served from the assets should count as a page view. */
export function isTrackable(request: Request, response: Response): boolean {
  if (request.method !== "GET") return false;
  // 304: the deployed HTML is must-revalidate, so every repeat visit is a 304.
  if (response.status !== 200 && response.status !== 304) return false;
  const { pathname } = new URL(request.url);
  if (pathname.startsWith("/_next/")) return false;
  if (pathname !== "/resume.pdf" && !isDocumentRequest(request)) return false;
  if (isPrefetch(request)) return false;
  return !isBot(request);
}

/** The page view record for a document load. */
export function pageViewFrom(request: Request): PageView {
  const url = new URL(request.url);
  return {
    url: url.href,
    hostname: url.hostname,
    referrer: request.headers.get("Referer"),
    userAgent: request.headers.get("User-Agent") ?? "",
    ip: request.headers.get("CF-Connecting-IP") ?? "",
    language: firstLanguage(request.headers.get("Accept-Language")),
  };
}

/**
 * Serves the asset exactly as before, then reports the view in the background.
 * Invariant: nothing in here can change or delay the response. All tracking
 * code runs inside the try/catch after the asset has been fetched.
 */
export async function handleRequest(
  request: Request,
  env: Env,
  ctx: ExecutionContext,
  provider: Provider,
): Promise<Response> {
  const response = await env.ASSETS.fetch(request);
  try {
    if (isTrackable(request, response)) {
      ctx.waitUntil(provider.track(pageViewFrom(request)));
    }
  } catch (error) {
    console.error("analytics: tracking failed", error);
  }
  return response;
}

export default {
  async fetch(request, env, ctx) {
    return handleRequest(request, env, ctx, createProvider(env));
  },
} satisfies ExportedHandler<Env>;
```

```ts
/** Short and unremarkable on purpose: generic blocker rules match words like "track" or "collect". */
export const PING_PATH = "/pv";

const MAX_PING_FIELD = 2000;

export type Ping = { url: string; referrer: string | null };

/** Validates a ping body. `url` must be a site-relative path; `referrer` is optional. */
export function parsePing(body: unknown): Ping | null {
  if (typeof body !== "object" || body === null) return null;
  const { url, referrer } = body as Record<string, unknown>;
  if (typeof url !== "string" || !url.startsWith("/") || url.startsWith("//")) return null;
  if (url.length > MAX_PING_FIELD) return null;
  if (referrer !== undefined && referrer !== null) {
    if (typeof referrer !== "string" || referrer.length > MAX_PING_FIELD) return null;
  }
  return { url, referrer: typeof referrer === "string" && referrer !== "" ? referrer : null };
}

/** The page view record for a client-side route change reported by the page. */
export function pageViewFromPing(request: Request, ping: Ping): PageView {
  const url = new URL(ping.url, new URL(request.url).origin);
  return { ...pageViewFrom(request), url: url.href, referrer: ping.referrer };
}
```

```tsx
"use client";

import { usePathname } from "next/navigation";
import { useEffect, useRef } from "react";

/** Must match PING_PATH in worker/index.ts. */
const PING_PATH = "/pv";

/**
 * Reports client-side route changes to the Worker. The initial document load
 * is counted at the edge, so the first render only remembers where we are.
 */
export function RouteTracker() {
  const pathname = usePathname();
  const last = useRef<{ pathname: string; href: string } | null>(null);

  useEffect(() => {
    if (last.current === null) {
      last.current = { pathname, href: window.location.href };
      return;
    }
    if (last.current.pathname === pathname) return;

    const referrer = last.current.href;
    last.current = { pathname, href: window.location.href };
    const body = JSON.stringify({ url: pathname + window.location.search, referrer });
    fetch(PING_PATH, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body,
      keepalive: true,
    }).catch(() => {
      // Analytics never surfaces errors to the visitor.
    });
  }, [pathname]);

  return null;
}
```

```ts
/** The site link in a resume export, tagged so a click from the resume shows up as such in analytics. */
function taggedSiteHref(url: string, medium: ResumeMedium): string {
  return `${url.replace(/\/$/, "")}/?utm_source=resume&utm_medium=${medium}`;
}

/** The resume exports' contact line, in display order. Absent links are omitted. */
export function contactItems(site: SiteInfo, medium: ResumeMedium): ContactItem[] {
  const { email, github, linkedin } = site.links;
  const items: ContactItem[] = [{ text: site.location }];
  if (email) items.push({ text: email, href: `mailto:${email}` });
  items.push({ text: displayUrl(site.url), href: taggedSiteHref(site.url, medium) });
  for (const url of [github, linkedin]) {
    if (url) items.push({ text: displayUrl(url), href: url });
  }
  return items;
}
```
